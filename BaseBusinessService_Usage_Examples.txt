// ========================================
// 🔥 CÁCH SỬ DỤNG BaseBusinessService THỰC TẾ
// ========================================

// 1. ✅ API Call: Lấy tất cả roles
// GET /api/Role/GetAll
// -> RoleController.GetAll() -> RoleService.GetAll() -> Return List<RoleDTO>

// 2. ✅ API Call: Lấy roles có phân trang 
// GET /api/Role/GetPaging?page=1&pageSize=10&searchTerm=admin
// -> RoleController.GetPaging() -> RoleService.GetPaging() -> Return paged data

// 3. ✅ API Call: Lấy role theo ID
// GET /api/Role/GetById/1
// -> RoleController.GetById(1) -> RoleService.GetById(1) -> Return RoleDTO

// 4. ✅ API Call: Thêm role mới
// POST /api/Role/SubmitData
// Body: {
//   "id": 0,
//   "roleName": "New Role",
//   "roleDescription": "Description",
//   "state": 1  // EntityState.Add
// }
// -> RoleController.SubmitData() -> RoleService.SubmitData() -> Add new role

// 5. ✅ API Call: Sửa role
// POST /api/Role/SubmitData  
// Body: {
//   "id": 1,
//   "roleName": "Updated Role", 
//   "roleDescription": "Updated Description",
//   "state": 2  // EntityState.Edit
// }
// -> RoleController.SubmitData() -> RoleService.SubmitData() -> Update role

// 6. ✅ API Call: Xóa role
// POST /api/Role/SubmitData
// Body: {
//   "id": 1,
//   "state": 3  // EntityState.Remove  
// }
// -> RoleController.SubmitData() -> RoleService.SubmitData() -> Soft delete role

// 7. ✅ API Call: Xóa trực tiếp
// DELETE /api/Role/1
// -> RoleController.Delete(1) -> RoleService.Delete(1) -> Soft delete

// ========================================
// 🎯 WORKFLOW THỰC TẾ:
// ========================================

// Client -> API Controller -> BaseBussinessController -> IRoleService -> BaseBusinessService -> Repository -> Database

// ========================================  
// 🚀 VÍ DỤ CODE FRONTEND:
// ========================================

// JavaScript fetch calls:

// Lấy tất cả roles
fetch('/api/Role/GetAll')
  .then(response => response.json())
  .then(data => console.log(data));

// Lấy roles có phân trang
fetch('/api/Role/GetPaging?page=1&pageSize=10&searchTerm=admin')
  .then(response => response.json()) 
  .then(data => console.log(data));

// Thêm role mới
fetch('/api/Role/SubmitData', {
  method: 'POST',
  headers: {'Content-Type': 'application/json'},
  body: JSON.stringify({
    id: 0,
    roleName: 'Manager',
    roleDescription: 'Manager role',
    state: 1 // Add
  })
})
.then(response => response.json())
.then(data => console.log(data));

// Sửa role
fetch('/api/Role/SubmitData', {
  method: 'POST', 
  headers: {'Content-Type': 'application/json'},
  body: JSON.stringify({
    id: 1,
    roleName: 'Senior Manager',
    roleDescription: 'Updated description', 
    state: 2 // Edit
  })
})
.then(response => response.json())
.then(data => console.log(data));

// Xóa role  
fetch('/api/Role/SubmitData', {
  method: 'POST',
  headers: {'Content-Type': 'application/json'},
  body: JSON.stringify({
    id: 1,
    state: 3 // Remove
  })
})
.then(response => response.json())
.then(data => console.log(data));
